function h($s) { return htmlspecialchars($s, ENT_QUOTES, 'UTF-8'); }
function real($p) {
    $p = str_replace('\\', '/', $p);
    while (strpos($p, '//') !== false) $p = str_replace('//', '/', $p);
    $p = str_replace('/./', '/', $p);
    $parts = explode('/', $p);
    $res = [];
    foreach ($parts as $part) {
        if ($part === '..') array_pop($res);
        elseif ($part !== '' && $part !== '.') $res[] = $part;
    }
    return '/' . implode('/', $res);
}
function size_h($b) {
    if ($b === null) return '-';
    if ($b >= 1073741824) return round($b/1073741824,1).' GB';
    if ($b >= 1048576) return round($b/1048576,1).' MB';
    if ($b >= 1024) return round($b/1024,1).' KB';
    return $b.' B';
}

$dir = isset($_GET['d']) ? real('/'.trim($_GET['d'],'/')) : '/';
$full = ROOT . $dir;
if (!is_dir($full)) { $dir = '/'; $full = ROOT; }

$msg = '';
$msg_t = '';

if ($auth && $auth === PASS) {
    // --- Create Folder ---
    if (isset($_POST['mkdir']) && !empty($_POST['dirname'])) {
        $nd = real($dir.'/'.basename($_POST['dirname']));
        $nf = ROOT.$nd;
        if (!file_exists($nf)) { mkdir($nf, 0755, true); $msg = "Folder created: ".h($nd); $msg_t='ok'; }
        else { $msg = "Already exists"; $msg_t='err'; }
    }
    // --- Create File ---
    if (isset($_POST['mkfile']) && !empty($_POST['filename'])) {
        $nd = real($dir.'/'.basename($_POST['filename']));
        $nf = ROOT.$nd;
        if (!file_exists($nf)) { file_put_contents($nf, ''); $msg = "File created: ".h($nd); $msg_t='ok'; }
        else { $msg = "Already exists"; $msg_t='err'; }
    }
    // --- Upload ---
    if (isset($_FILES['upfile']) && $_FILES['upfile']['error'] === UPLOAD_ERR_OK) {
        $dest = $full.'/'.basename($_FILES['upfile']['name']);
        if (move_uploaded_file($_FILES['upfile']['tmp_name'], $dest)) {
            $msg = "Uploaded: ".h($_FILES['upfile']['name']); $msg_t='ok';
        } else { $msg = "Upload failed"; $msg_t='err'; }
    }
    // --- Delete ---
    if (isset($_GET['del'])) {
        $target = real($dir.'/'.basename($_GET['del']));
        $tf = ROOT.$target;
        if ($tf !== ROOT.'/' && file_exists($tf)) {
            if (is_dir($tf)) {
                $it = new RecursiveDirectoryIterator($tf, RecursiveDirectoryIterator::SKIP_DOTS);
                $files = new RecursiveIteratorIterator($it, RecursiveIteratorIterator::CHILD_FIRST);
                foreach ($files as $f) { $f->isDir() ? rmdir($f->getRealPath()) : unlink($f->getRealPath()); }
                rmdir($tf);
            } else { unlink($tf); }
            $msg = "Deleted: ".h($target); $msg_t='ok';
        }
    }
    // --- Rename ---
    if (isset($_POST['rename'], $_POST['newname']) && !empty($_POST['newname'])) {
        $old = real($dir.'/'.basename($_POST['rename']));
        $new = real($dir.'/'.basename($_POST['newname']));
        if (file_exists(ROOT.$old) && !file_exists(ROOT.$new)) {
            rename(ROOT.$old, ROOT.$new); $msg = "Renamed"; $msg_t='ok';
        } else { $msg = "Rename failed"; $msg_t='err'; }
    }
    // --- Save file ---
    if (isset($_POST['savefile'], $_POST['content'])) {
        $sf = real($dir.'/'.basename($_POST['savefile']));
        file_put_contents(ROOT.$sf, $_POST['content']);
        $msg = "Saved: ".h($sf); $msg_t='ok';
    }
}
